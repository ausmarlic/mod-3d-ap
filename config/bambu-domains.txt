# Domains whose resolved IPs AP clients may reach when UPLINK_MODE=bambu.

# Installed copy: /etc/mod-3d-ap/bambu-domains.txt (re-run install.sh to update).
#
# Source: https://wiki.bambulab.com/en/general/printer-network-ports plus
# OpenBambuAPI. Bambu says the list is not exhaustive and changes over time.

bambulab.com
bambulab.cn
bblmw.com
# MQTT broker CNAME target (unverified)
emqx.cloud

# Object storage for files/firmware. NOTE: these are broad (any S3 / Aliyun OSS

amazonaws.com
aliyuncs.com

# Remote camera (TUTK/Kalay). Best-effort only: media relays are handed out as
